## ライブラリの追加

Nordic提供の **nRF Connect Device Manager** を Swift Package Manager で追加します。

```
1. Xcode → File → Add Package Dependencies
2. URL: https://github.com/NordicSemiconductor/IOS-nRF-Connect-Device-Manager
3. バージョン: latest (1.8+)
4. 追加: iOSMcuManagerLibrary
```


## DFUマネージャの実装

```swift
import SwiftUI
import iOSMcuManagerLibrary

@Observable
final class DFUManager: NSObject {

    var progress: Double = 0.0        // 0.0〜1.0
    var status = "待機中"
    var isUpgrading = false
    var isComplete = false
    var errorMessage: String?

    private var peripheral: CBPeripheral?
    private var transport: McuMgrBleTransport?
    private var upgradeManager: FirmwareUpgradeManager?

    // ペリフェラルをセット（BLEスキャン後に呼ぶ）
    func setPeripheral(_ peripheral: CBPeripheral) {
        self.peripheral = peripheral
        self.transport = McuMgrBleTransport(peripheral)
    }

    // DFU開始
    func startDFU(with zipFileURL: URL) {
        guard let transport else {
            errorMessage = "デバイス未接続"
            return
        }

        isUpgrading = true
        isComplete = false
        progress = 0.0
        status = "DFUパッケージを読み込み中..."

        // ZIPパッケージを読み込む
        do {
            let package = try McuMgrPackage(from: zipFileURL)
            let config = FirmwareUpgradeConfiguration(
                estimatedSwapTime: 10,          // Swap完了予想時間 (秒)
                pipelineDepth: 3,               // パイプライン深度（並列Write数）
                byteAlignment: ImageUploadAlignment.fourByte
            )
            upgradeManager = FirmwareUpgradeManager(transport: transport, delegate: self)
            upgradeManager?.configuration = config
            try upgradeManager?.start(package: package)
            status = "DFU転送中..."
        } catch {
            isUpgrading = false
            errorMessage = "パッケージ読み込みエラー: \(error.localizedDescription)"
        }
    }

    // DFU中断
    func cancel() {
        upgradeManager?.cancel()
        isUpgrading = false
        status = "キャンセル"
    }
}
```


## FirmwareUpgradeDelegate の実装

```swift
extension DFUManager: FirmwareUpgradeDelegate {

    // DFU開始
    func upgradeDidStart(controller: FirmwareUpgradeController) {
        status = "DFU開始"
    }

    // 進捗更新
    func upgradeStateDidChange(from previousState: FirmwareUpgradeState,
                               to newState: FirmwareUpgradeState) {
        switch newState {
        case .requestMcuMgrParameters: status = "MCU Manager パラメータ取得中..."
        case .uploadingImages:         status = "ファームウェア転送中..."
        case .swappingImages:          status = "イメージ書き換え中..."
        case .confirmingImage:         status = "イメージ確認中..."
        case .success:                 status = "完了"
        default:                       break
        }
    }

    // 転送進捗
    func upgradeDidFail(inState state: FirmwareUpgradeState, with error: Error) {
        isUpgrading = false
        errorMessage = "エラー (\(state)): \(error.localizedDescription)"
        status = "失敗"
    }

    func upgradeDidCancel(state: FirmwareUpgradeState) {
        isUpgrading = false
        status = "キャンセル"
    }

    func uploadProgressDidChange(bytesSent: Int, imageSize: Int, timestamp: Date) {
        progress = Double(bytesSent) / Double(imageSize)
        let percent = Int(progress * 100)
        status = "転送中... \(percent)% (\(bytesSent)/\(imageSize) bytes)"
    }

    func upgradeDidComplete() {
        isUpgrading = false
        isComplete = true
        progress = 1.0
        status = "DFU完了! デバイスが再起動中..."
    }
}
```


## DFU進捗表示UI

```swift
struct DFUView: View {
    @State private var dfuManager = DFUManager()
    @State private var showFilePicker = false

    var body: some View {
        VStack(spacing: 28) {
            Text("OTA ファームウェア更新")
                .font(.title2).bold()

            // 進捗バー
            VStack(spacing: 8) {
                ProgressView(value: dfuManager.progress)
                    .progressViewStyle(.linear)
                    .tint(dfuManager.isComplete ? .green : .blue)
                    .animation(.easeInOut, value: dfuManager.progress)

                Text(dfuManager.status)
                    .font(.caption)
                    .foregroundStyle(.secondary)
            }

            // パーセント表示
            Text("\(Int(dfuManager.progress * 100))%")
                .font(.system(size: 48, weight: .bold, design: .rounded))
                .foregroundStyle(dfuManager.isComplete ? .green : .primary)
                .contentTransition(.numericText())

            // アクションボタン
            if !dfuManager.isUpgrading {
                Button("DFUパッケージを選択") {
                    showFilePicker = true
                }
                .buttonStyle(.borderedProminent)
                .disabled(dfuManager.isUpgrading)
            } else {
                Button("キャンセル", role: .destructive) {
                    dfuManager.cancel()
                }
                .buttonStyle(.bordered)
            }

            // エラー表示
            if let error = dfuManager.errorMessage {
                Text(error)
                    .foregroundStyle(.red)
                    .font(.caption)
                    .multilineTextAlignment(.center)
            }
        }
        .padding(32)
        .fileImporter(
            isPresented: $showFilePicker,
            allowedContentTypes: [.zip],
            allowsMultipleSelection: false
        ) { result in
            if case .success(let urls) = result, let url = urls.first {
                url.startAccessingSecurityScopedResource()
                dfuManager.startDFU(with: url)
            }
        }
    }
}
```


## MTUの事前ネゴシエーション

転送速度を最大化するため、DFU開始前にMTUを交換します。`McuMgrBleTransport` が内部で行いますが、接続パラメータを調整しておくと効果的です。

```swift
// ペリフェラル接続後に MTU を最大まで要求
centralManager.connect(peripheral, options: [
    CBConnectPeripheralOptionNotifyOnDisconnectionKey: true
])

// delegate で接続直後に
func centralManager(_ central: CBCentralManager, didConnect peripheral: CBPeripheral) {
    // McuMgrBleTransport が自動でMTUネゴシエーションを行う
    dfuManager.setPeripheral(peripheral)
}
```


**まとめ**

- **nRF Connect Device Manager** ライブラリで SMP over BLE の複雑な処理が抽象化される
- `FirmwareUpgradeManager.start(package:)` でDFUを開始、`FirmwareUpgradeDelegate` で進捗を受け取る
- `uploadProgressDidChange` で進捗バーを更新、`upgradeDidComplete` で完了通知
- MTUネゴシエーションはライブラリが自動実行
