## センサー読み出しコード

```c
#include <zephyr/kernel.h>
#include <zephyr/device.h>
#include <zephyr/drivers/sensor.h>
#include <zephyr/logging/log.h>

LOG_MODULE_REGISTER(sensor, LOG_LEVEL_INF);

/* BME280デバイス参照 */
static const struct device *bme280;

/* センサーデータ構造体 */
struct sensor_data {
    int16_t  temp_hundredths;   /* 温度 × 100 (°C) */
    uint16_t hum_hundredths;    /* 湿度 × 100 (%) */
    uint32_t press_pa;          /* 気圧 (Pa) */
};

static struct sensor_data latest_data;

/* センサーの初期化 */
int sensor_init(void)
{
    bme280 = DEVICE_DT_GET_ANY(bosch_bme280);
    if (!device_is_ready(bme280)) {
        LOG_ERR("BME280 デバイスが見つかりません");
        return -ENODEV;
    }
    LOG_INF("BME280 初期化完了: %s", bme280->name);
    return 0;
}

/* センサー読み出し */
int sensor_read(struct sensor_data *out)
{
    struct sensor_value temp, hum, press;
    int ret;

    /* センサーからデータ取得 */
    ret = sensor_sample_fetch(bme280);
    if (ret) {
        LOG_ERR("サンプル取得失敗: %d", ret);
        return ret;
    }

    /* 各チャンネルの値を取得 */
    sensor_channel_get(bme280, SENSOR_CHAN_AMBIENT_TEMP, &temp);
    sensor_channel_get(bme280, SENSOR_CHAN_HUMIDITY, &hum);
    sensor_channel_get(bme280, SENSOR_CHAN_PRESS, &press);

    /*
     * sensor_value: val1 = 整数部, val2 = 小数部 (百万分の1)
     * 例: 24.56°C → val1=24, val2=560000
     * → hundredths: 24*100 + 560000/10000 = 2456
     */
    out->temp_hundredths = (int16_t)(temp.val1 * 100 + temp.val2 / 10000);
    out->hum_hundredths  = (uint16_t)(hum.val1 * 100 + hum.val2 / 10000);
    /* 気圧: kPa → Pa に変換 */
    out->press_pa = (uint32_t)(press.val1 * 1000 + press.val2 / 1000);

    LOG_INF("温度: %d.%02d°C  湿度: %d.%02d%%  気圧: %dPa",
            out->temp_hundredths / 100, out->temp_hundredths % 100,
            out->hum_hundredths  / 100, out->hum_hundredths  % 100,
            out->press_pa);

    return 0;
}
```


## 定期測定ワークハンドラ

1秒ごとにセンサーを読み取り、GATTへ通知するワークキュータスクを作成します。GATTとの連携は次節で実装します。

```c
#include <zephyr/kernel.h>

static struct k_work_delayable sensor_work;

/* ワークハンドラ: 1秒ごとに呼ばれる */
static void sensor_work_handler(struct k_work *work)
{
    struct sensor_data data;

    if (sensor_read(&data) == 0) {
        /* latest_dataを更新し、BLE Notify関数を呼ぶ */
        latest_data = data;
        ble_notify_sensor_data(&data);   /* 次節で実装 */
    }

    /* 1秒後に再スケジュール */
    k_work_reschedule(&sensor_work, K_SECONDS(1));
}

void sensor_start_periodic(void)
{
    k_work_init_delayable(&sensor_work, sensor_work_handler);
    k_work_reschedule(&sensor_work, K_SECONDS(1));
}
```


## sensor_value の符号考慮

温度が負（冷蔵庫など）のケースでは `val1` が負になります。

```c
/* 負温度の例: -3.50°C → val1=-3, val2=-500000 */
/* hundredths = (-3)*100 + (-500000/10000) = -300 + (-50) = -350 */
out->temp_hundredths = (int16_t)(temp.val1 * 100
                       + (temp.val2 < 0 ? -1 : 1) * (abs(temp.val2) / 10000));
```

小数部 `val2` は `val1` と同符号です。


**まとめ**

- `DEVICE_DT_GET_ANY(bosch_bme280)` でデバイスツリーからBME280デバイスを取得
- `sensor_sample_fetch` → `sensor_channel_get` で各チャンネルを読み出す
- `sensor_value` の val1/val2 を `× 100` でint16_tに変換して送信
- `k_work_delayable` で1秒ごとの定期測定を実装

次節ではGATTサービスの定義とNotify送信・Writeハンドラを実装します。
