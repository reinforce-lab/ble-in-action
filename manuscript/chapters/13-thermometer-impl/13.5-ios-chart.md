## Swift Charts による温度グラフ

iOS 16以降で使える **Swift Charts** フレームワークを使います。宣言的なAPIでアニメーション付きのグラフが簡単に作れます。

```swift
import SwiftUI
import Charts

struct ThermometerView: View {
    @State private var bleManager = ThermometerBLEManager()
    @State private var thresholdInput = ""
    @State private var showThresholdSheet = false

    var body: some View {
        NavigationStack {
            ScrollView {
                VStack(spacing: 24) {
                    // 大きな現在値表示
                    CurrentReadingView(
                        temp: bleManager.latestTemp,
                        hum:  bleManager.latestHum,
                        isConnected: bleManager.isConnected
                    )

                    // 温度グラフ
                    TemperatureChartView(
                        readings: bleManager.readings,
                        threshold: bleManager.alertThreshold
                    )

                    // しきい値設定
                    ThresholdControlView(
                        threshold: bleManager.alertThreshold,
                        onUpdate: { bleManager.writeThreshold($0) }
                    )
                }
                .padding()
            }
            .navigationTitle("BLE温度計")
            .toolbar {
                ToolbarItem(placement: .navigationBarTrailing) {
                    StatusBadge(message: bleManager.statusMessage,
                                connected: bleManager.isConnected)
                }
            }
        }
    }
}
```

## 温度グラフの実装

```swift
struct TemperatureChartView: View {
    let readings: [SensorReading]
    let threshold: Double

    var body: some View {
        VStack(alignment: .leading, spacing: 8) {
            Text("温度グラフ（直近2分間）")
                .font(.headline)

            if readings.isEmpty {
                Text("データ待機中...")
                    .foregroundStyle(.secondary)
                    .frame(height: 200)
                    .frame(maxWidth: .infinity)
            } else {
                Chart {
                    // 温度折れ線グラフ
                    ForEach(readings) { r in
                        LineMark(
                            x: .value("時刻", r.timestamp),
                            y: .value("温度", r.temperature)
                        )
                        .foregroundStyle(.red)
                        .interpolationMethod(.catmullRom)  // 滑らかな曲線
                    }

                    // 最新値の点
                    if let last = readings.last {
                        PointMark(
                            x: .value("時刻", last.timestamp),
                            y: .value("温度", last.temperature)
                        )
                        .foregroundStyle(.red)
                        .symbolSize(60)
                    }

                    // しきい値ライン
                    RuleMark(y: .value("しきい値", threshold))
                        .foregroundStyle(.orange.opacity(0.8))
                        .lineStyle(StrokeStyle(lineWidth: 1.5, dash: [6, 3]))
                        .annotation(position: .top, alignment: .trailing) {
                            Text("\(threshold, specifier: "%.1f")°C")
                                .font(.caption)
                                .foregroundStyle(.orange)
                        }
                }
                .chartYScale(domain: yDomain)
                .chartXAxis {
                    AxisMarks(values: .stride(by: .second, count: 30)) { _ in
                        AxisGridLine()
                        AxisTick()
                        AxisValueLabel(format: .dateTime.minute().second())
                    }
                }
                .chartYAxis {
                    AxisMarks(position: .leading)
                }
                .frame(height: 200)
                .animation(.easeInOut(duration: 0.3), value: readings.count)
            }
        }
        .padding()
        .background(RoundedRectangle(cornerRadius: 12).fill(.ultraThinMaterial))
    }

    // Y軸範囲: 最小〜最大 + マージン
    private var yDomain: ClosedRange<Double> {
        let temps = readings.map(\.temperature)
        let min = (temps.min() ?? 15.0) - 2.0
        let max = (temps.max() ?? 35.0) + 2.0
        return min...max
    }
}
```

## 現在値ビュー

```swift
struct CurrentReadingView: View {
    let temp: Double
    let hum: Double
    let isConnected: Bool

    var tempColor: Color {
        temp > 30 ? .red : temp > 25 ? .orange : .blue
    }

    var body: some View {
        HStack(spacing: 32) {
            VStack {
                Text(String(format: "%.1f", temp))
                    .font(.system(size: 56, weight: .bold, design: .rounded))
                    .foregroundStyle(tempColor)
                    .contentTransition(.numericText())
                Text("°C")
                    .font(.title3)
                    .foregroundStyle(.secondary)
            }
            Divider().frame(height: 60)
            VStack {
                Text(String(format: "%.1f", hum))
                    .font(.system(size: 56, weight: .bold, design: .rounded))
                    .foregroundStyle(.cyan)
                    .contentTransition(.numericText())
                Text("%")
                    .font(.title3)
                    .foregroundStyle(.secondary)
            }
        }
        .padding()
        .background(RoundedRectangle(cornerRadius: 16).fill(.ultraThinMaterial))
        .overlay(alignment: .topTrailing) {
            Circle()
                .fill(isConnected ? .green : .red)
                .frame(width: 10, height: 10)
                .padding(8)
        }
    }
}
```

## しきい値設定コンポーネント

```swift
struct ThresholdControlView: View {
    let threshold: Double
    let onUpdate: (Double) -> Void
    @State private var value: Double = 30.0

    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            Text("アラートしきい値")
                .font(.headline)
            HStack {
                Slider(value: $value, in: 0...50, step: 0.5)
                    .tint(.orange)
                    .onChange(of: value) { _, new in
                        onUpdate(new)
                    }
                Text(String(format: "%.1f°C", value))
                    .font(.system(.body, design: .monospaced))
                    .frame(width: 70, alignment: .trailing)
            }
        }
        .padding()
        .background(RoundedRectangle(cornerRadius: 12).fill(.ultraThinMaterial))
        .onAppear { value = threshold }
        .onChange(of: threshold) { _, new in value = new }
    }
}
```

---

**まとめ**

- `Swift Charts` の `LineMark` + `RuleMark` でしきい値ライン付き折れ線グラフ
- `chartYScale(domain:)` でY軸範囲を動的に計算
- `.contentTransition(.numericText())` で現在値のアニメーション更新
- `Slider` でしきい値を調整 → `bleManager.writeThreshold()` でWrite

次節でシステム全体のまとめと動作確認を行います。
