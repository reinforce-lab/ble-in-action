## データモデルとBLEマネージャ

```swift
import CoreBluetooth
import Observation

// MARK: - UUIDs
extension CBUUID {
    static let sensorService   = CBUUID(string: "12345678-1234-5678-1234-56789abcdef0")
    static let tempNotify      = CBUUID(string: "12345678-1234-5678-1234-56789abcdef2")
    static let alertThreshold  = CBUUID(string: "12345678-1234-5678-1234-56789abcdef3")
}

// MARK: - データモデル
struct SensorReading: Identifiable {
    let id = UUID()
    let timestamp: Date
    let temperature: Double   // °C
    let humidity: Double      // %
}

// MARK: - BLE管理クラス
@Observable
final class ThermometerBLEManager: NSObject {

    var readings: [SensorReading] = []
    var latestTemp: Double = 0
    var latestHum:  Double = 0
    var alertThreshold: Double = 30.0
    var isConnected = false
    var statusMessage = "スキャン中..."

    private var centralManager: CBCentralManager!
    private var peripheral: CBPeripheral?
    private var thresholdChar: CBCharacteristic?

    override init() {
        super.init()
        centralManager = CBCentralManager(delegate: self, queue: nil)
    }

    // しきい値をセンサーに書き込む
    func writeThreshold(_ celsius: Double) {
        guard let char = thresholdChar,
              let periph = peripheral else { return }
        alertThreshold = celsius
        let hundredths = Int16(celsius * 100)
        var le = hundredths.littleEndian
        let data = Data(bytes: &le, count: 2)
        periph.writeValue(data, for: char, type: .withResponse)
    }
}
```


## CBCentralManagerDelegate

```swift
extension ThermometerBLEManager: CBCentralManagerDelegate {

    func centralManagerDidUpdateState(_ central: CBCentralManager) {
        if central.state == .poweredOn {
            central.scanForPeripherals(withServices: [.sensorService], options: nil)
            statusMessage = "センサーを検索中..."
        }
    }

    func centralManager(_ central: CBCentralManager,
                        didDiscover peripheral: CBPeripheral,
                        advertisementData: [String: Any],
                        rssi RSSI: NSNumber) {
        guard self.peripheral == nil else { return }
        self.peripheral = peripheral
        central.stopScan()
        central.connect(peripheral, options: nil)
        statusMessage = "接続中..."
    }

    func centralManager(_ central: CBCentralManager,
                        didConnect peripheral: CBPeripheral) {
        peripheral.delegate = self
        peripheral.discoverServices([.sensorService])
        isConnected = true
        statusMessage = "接続済み"
    }

    func centralManager(_ central: CBCentralManager,
                        didDisconnectPeripheral peripheral: CBPeripheral,
                        error: Error?) {
        self.peripheral = nil
        thresholdChar = nil
        isConnected = false
        statusMessage = "切断 — 再スキャン中..."
        // 自動再接続
        central.scanForPeripherals(withServices: [.sensorService], options: nil)
    }
}
```


## CBPeripheralDelegate

```swift
extension ThermometerBLEManager: CBPeripheralDelegate {

    func peripheral(_ peripheral: CBPeripheral,
                    didDiscoverServices error: Error?) {
        guard let services = peripheral.services else { return }
        for service in services where service.uuid == .sensorService {
            peripheral.discoverCharacteristics([.tempNotify, .alertThreshold],
                                               for: service)
        }
    }

    func peripheral(_ peripheral: CBPeripheral,
                    didDiscoverCharacteristicsFor service: CBService,
                    error: Error?) {
        guard let chars = service.characteristics else { return }
        for char in chars {
            switch char.uuid {
            case .tempNotify:
                peripheral.setNotifyValue(true, for: char)
            case .alertThreshold:
                thresholdChar = char
                peripheral.readValue(for: char)   // 現在のしきい値を読む
            default:
                break
            }
        }
    }

    // Notifyデータ受信
    func peripheral(_ peripheral: CBPeripheral,
                    didUpdateValueFor characteristic: CBCharacteristic,
                    error: Error?) {
        guard error == nil, let data = characteristic.value else { return }

        switch characteristic.uuid {
        case .tempNotify:
            parseNotifyData(data)
        case .alertThreshold:
            parseThreshold(data)
        default:
            break
        }
    }

    // 4バイト Notify パース: Int16(温度×100) + UInt16(湿度×100)
    private func parseNotifyData(_ data: Data) {
        guard data.count >= 4 else { return }
        let tempHundredths = Int16(littleEndian: data.withUnsafeBytes { $0.load(fromByteOffset: 0, as: Int16.self) })
        let humHundredths  = UInt16(littleEndian: data.withUnsafeBytes { $0.load(fromByteOffset: 2, as: UInt16.self) })

        let temp = Double(tempHundredths) / 100.0
        let hum  = Double(humHundredths)  / 100.0

        latestTemp = temp
        latestHum  = hum

        let reading = SensorReading(timestamp: Date(), temperature: temp, humidity: hum)
        readings.append(reading)
        // 直近120件（2分間）を保持
        if readings.count > 120 { readings.removeFirst() }
    }

    private func parseThreshold(_ data: Data) {
        guard data.count >= 2 else { return }
        let v = Int16(littleEndian: data.withUnsafeBytes { $0.load(as: Int16.self) })
        alertThreshold = Double(v) / 100.0
    }
}
```


**まとめ**

- `@Observable` クラスでBLEマネージャをSwiftUIと統合
- Notify受信: `Int16(温度×100)` + `UInt16(湿度×100)` を4バイトでパース
- しきい値はReadで初期値取得、Writeで更新
- 切断後は自動スキャン再開で常時接続を維持
