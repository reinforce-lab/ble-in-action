## GATTサービス定義

温度計サービスを `BT_GATT_SERVICE_DEFINE` マクロで宣言します。

```c
#include <zephyr/bluetooth/gatt.h>
#include <zephyr/sys/byteorder.h>

/* サービス/キャラクタリスティクス UUID */
#define SENSOR_SVC_UUID \
    BT_UUID_DECLARE_128(BT_UUID_128_ENCODE(0x12345678, 0x1234, 0x5678, 0x1234, 0x56789abcdef0))
#define TEMP_NOTIFY_UUID \
    BT_UUID_DECLARE_128(BT_UUID_128_ENCODE(0x12345678, 0x1234, 0x5678, 0x1234, 0x56789abcdef2))
#define THRESHOLD_UUID \
    BT_UUID_DECLARE_128(BT_UUID_128_ENCODE(0x12345678, 0x1234, 0x5678, 0x1234, 0x56789abcdef3))

/* しきい値 (デフォルト 30.00°C = 3000) */
static int16_t alert_threshold = 3000;

/* Notify有効化状態変化コールバック */
static void temp_ccc_changed(const struct bt_gatt_attr *attr, uint16_t value)
{
    bool notif_enabled = (value == BT_GATT_CCC_NOTIFY);
    printk("Temperature Notify: %s\n", notif_enabled ? "有効" : "無効");
}

/* Threshold Read ハンドラ */
static ssize_t threshold_read(struct bt_conn *conn,
                              const struct bt_gatt_attr *attr,
                              void *buf, uint16_t len, uint16_t offset)
{
    int16_t val = sys_cpu_to_le16(alert_threshold);
    return bt_gatt_attr_read(conn, attr, buf, len, offset, &val, sizeof(val));
}

/* Threshold Write ハンドラ */
static ssize_t threshold_write(struct bt_conn *conn,
                               const struct bt_gatt_attr *attr,
                               const void *buf, uint16_t len,
                               uint16_t offset, uint8_t flags)
{
    if (len != sizeof(int16_t)) {
        return BT_GATT_ERR(BT_ATT_ERR_INVALID_ATTRIBUTE_LEN);
    }
    int16_t new_threshold;
    memcpy(&new_threshold, buf, sizeof(new_threshold));
    alert_threshold = sys_le16_to_cpu(new_threshold);

    printk("しきい値更新: %d.%02d°C\n",
           alert_threshold / 100, abs(alert_threshold % 100));
    return len;
}

/* GATTサービス定義 */
BT_GATT_SERVICE_DEFINE(sensor_svc,
    /* プライマリサービス */
    BT_GATT_PRIMARY_SERVICE(SENSOR_SVC_UUID),

    /* 温度/湿度 Notifyキャラクタリスティクス */
    BT_GATT_CHARACTERISTIC(TEMP_NOTIFY_UUID,
                           BT_GATT_CHRC_NOTIFY,
                           BT_GATT_PERM_NONE,
                           NULL, NULL, NULL),
    BT_GATT_CCC(temp_ccc_changed, BT_GATT_PERM_READ | BT_GATT_PERM_WRITE),

    /* しきい値 Read/Writeキャラクタリスティクス */
    BT_GATT_CHARACTERISTIC(THRESHOLD_UUID,
                           BT_GATT_CHRC_READ | BT_GATT_CHRC_WRITE,
                           BT_GATT_PERM_READ | BT_GATT_PERM_WRITE,
                           threshold_read, threshold_write, &alert_threshold),
);
```

## Notify 送信

Notifyデータは 4バイトのペイロードにパックします。

```c
/* Notify ペイロード構造体 */
struct __packed sensor_notify_data {
    int16_t  temp_hundredths;   /* 温度 × 100 */
    uint16_t hum_hundredths;    /* 湿度 × 100 */
};

void ble_notify_sensor_data(const struct sensor_data *data)
{
    struct sensor_notify_data payload = {
        .temp_hundredths = sys_cpu_to_le16(data->temp_hundredths),
        .hum_hundredths  = sys_cpu_to_le16(data->hum_hundredths),
    };

    /* GATT属性インデックス: PRIMARY_SERVICEが[0], CHARACTERISTICが[1], 値が[2] */
    const struct bt_gatt_attr *attr = &sensor_svc.attrs[2];

    int ret = bt_gatt_notify(NULL, attr, &payload, sizeof(payload));
    if (ret < 0 && ret != -ENOTCONN) {
        printk("Notify送信失敗: %d\n", ret);
    }

    /* しきい値チェック */
    if (data->temp_hundredths >= alert_threshold) {
        printk("⚠ 温度アラート: %d.%02d°C（しきい値: %d.%02d°C）\n",
               data->temp_hundredths / 100, data->temp_hundredths % 100,
               alert_threshold / 100, alert_threshold % 100);
        /* アラート専用Notifyを送る場合はここで追加実装 */
    }
}
```

## main.c の全体フロー

```c
#include <zephyr/kernel.h>
#include <zephyr/bluetooth/bluetooth.h>

static const struct bt_data ad[] = {
    BT_DATA_BYTES(BT_DATA_FLAGS, BT_LE_AD_GENERAL | BT_LE_AD_NO_BREDR),
    BT_DATA_BYTES(BT_DATA_UUID128_ALL,
        BT_UUID_128_ENCODE(0x12345678, 0x1234, 0x5678, 0x1234, 0x56789abcdef0)),
};

int main(void)
{
    int err;

    /* センサー初期化 */
    err = sensor_init();
    if (err) return err;

    /* BLE初期化 */
    err = bt_enable(NULL);
    if (err) return err;

    /* アドバタイズ開始 */
    err = bt_le_adv_start(BT_LE_ADV_CONN_NAME, ad, ARRAY_SIZE(ad), NULL, 0);
    if (err) {
        printk("アドバタイズ開始失敗: %d\n", err);
        return err;
    }

    /* 定期センサー読み出し開始 */
    sensor_start_periodic();

    printk("温度計 起動完了\n");
    return 0;
}
```

**まとめ**

- `BT_GATT_SERVICE_DEFINE` でNotify + Read/Writeキャラクタリスティクスを宣言
- `BT_GATT_CCC` マクロでCCCDを追加し、Notify有効化コールバックを設定
- `bt_gatt_notify()` でペイロードを送信。`-ENOTCONN` は接続なし（正常）
- しきい値超過時に追加Notifyやログを送れるよう実装
