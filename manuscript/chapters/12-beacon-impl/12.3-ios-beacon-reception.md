## Core Location でのビーコン利用

iOSでiBeaconを受信するには `Core Bluetooth` ではなく **`Core Location`** フレームワークを使います。iBeaconはApple独自フォーマットのためOSがハードウェアレベルで処理し、バックグラウンドでも動作します。

```swift
import CoreLocation

class BeaconManager: NSObject, CLLocationManagerDelegate {

    private let locationManager = CLLocationManager()

    // 本書iBeaconのUUID
    private let beaconUUID = UUID(uuidString: "01234567-89ab-cdef-fedc-ba9876543210")!

    override init() {
        super.init()
        locationManager.delegate = self
        // 位置情報許可のリクエスト（iBeaconにも必要）
        locationManager.requestAlwaysAuthorization()
    }

    // ビーコンのモニタリング開始（範囲 in/out の検知）
    func startMonitoring() {
        let constraint = CLBeaconIdentityConstraint(uuid: beaconUUID)
        let region = CLBeaconRegion(beaconIdentityConstraint: constraint,
                                   identifier: "com.example.blebeacon")
        region.notifyOnEntry = true
        region.notifyOnExit  = true
        locationManager.startMonitoring(for: region)
        print("モニタリング開始")
    }

    // ビーコンのレンジング開始（詳細な距離情報）
    func startRanging() {
        let constraint = CLBeaconIdentityConstraint(uuid: beaconUUID)
        locationManager.startRangingBeacons(satisfying: constraint)
        print("レンジング開始")
    }
}
```


## モニタリングとレンジングの違い

| 機能 | モニタリング | レンジング |
|---|---|---|
| 目的 | 領域の入退出を検知 | 距離・RSSI・近接を取得 |
| 精度 | 低（in/out のみ） | 高（即時更新） |
| バックグラウンド | 可（アプリ非起動でも） | フォアグラウンドのみ |
| 電力消費 | 低 | 高 |
| デリゲート | `didEnterRegion` / `didExitRegion` | `didRangeBeacons` |


## デリゲートメソッドの実装

```swift
extension BeaconManager {

    // --- モニタリング ---

    // ビーコン領域に入った
    func locationManager(_ manager: CLLocationManager,
                         didEnterRegion region: CLRegion) {
        guard let beaconRegion = region as? CLBeaconRegion else { return }
        print("ビーコン領域に入りました: \(beaconRegion.identifier)")
        // アプリがバックグラウンドでも呼ばれる → ローカル通知を発行
        sendNotification(title: "センサーエリアに入りました")
        // レンジングで詳細確認
        startRanging()
    }

    // ビーコン領域から出た
    func locationManager(_ manager: CLLocationManager,
                         didExitRegion region: CLRegion) {
        print("ビーコン領域を出ました")
        sendNotification(title: "センサーエリアを出ました")
    }

    // --- レンジング ---

    // レンジング結果（1秒ごとに更新）
    func locationManager(_ manager: CLLocationManager,
                         didRangeBeacons beacons: [CLBeacon],
                         satisfying constraint: CLBeaconIdentityConstraint) {
        for beacon in beacons {
            let proximity = proximityString(beacon.proximity)
            let rssi      = beacon.rssi
            let accuracy  = String(format: "%.2f", beacon.accuracy)
            print("""
                UUID: \(beacon.uuid)
                Major: \(beacon.major)  Minor: \(beacon.minor)
                近接: \(proximity)  RSSI: \(rssi) dBm  距離: \(accuracy)m
            """)
            updateUI(proximity: beacon.proximity, distance: beacon.accuracy)
        }
    }

    private func proximityString(_ proximity: CLProximity) -> String {
        switch proximity {
        case .immediate: return "即近 (< 0.5m)"
        case .near:      return "近 (0.5〜3m)"
        case .far:       return "遠 (3m〜)"
        case .unknown:   return "不明"
        @unknown default: return "不明"
        }
    }
}
```


## SwiftUIでの表示

```swift
import SwiftUI
import CoreLocation

struct BeaconView: View {
    @StateObject private var viewModel = BeaconViewModel()

    var body: some View {
        VStack(spacing: 24) {
            Text("ビーコン近接検知")
                .font(.title2).bold()

            ProximityIndicator(proximity: viewModel.proximity)

            VStack(alignment: .leading, spacing: 8) {
                Text("RSSI: \(viewModel.rssi) dBm")
                Text("距離: \(String(format: "%.1f", viewModel.distance)) m")
                Text("Major: \(viewModel.major)  Minor: \(viewModel.minor)")
            }
            .font(.system(.body, design: .monospaced))

            Button("モニタリング開始") { viewModel.start() }
                .buttonStyle(.borderedProminent)
        }
        .padding()
    }
}

struct ProximityIndicator: View {
    let proximity: CLProximity

    var color: Color {
        switch proximity {
        case .immediate: return .green
        case .near:      return .yellow
        case .far:       return .orange
        default:         return .gray
        }
    }

    var body: some View {
        Circle()
            .fill(color)
            .frame(width: 80, height: 80)
            .overlay(
                Text(proximity == .immediate ? "至近" :
                     proximity == .near ? "近い" :
                     proximity == .far  ? "遠い" : "?")
                    .bold()
            )
    }
}
```


## Info.plist の必要キー

```xml
<key>NSLocationAlwaysAndWhenInUseUsageDescription</key>
<string>ビーコンの検知のために位置情報を使用します</string>
<key>NSLocationWhenInUseUsageDescription</key>
<string>ビーコンの検知のために位置情報を使用します</string>
```

バックグラウンドでのモニタリングには `Always` 許可が必要です。


**まとめ**

- iOSでのiBeacon受信は `Core Location` の `CLBeaconRegion` を使う
- モニタリング（in/out）はバックグラウンドでも動作する
- レンジング（距離・近接値）はフォアグラウンドのみ
- `CLProximity` で immediate / near / far の近接段階を取得できる
