% ====================================================================
% BLEInAction – LaTeX Preamble (LuaLaTeX)
% ====================================================================
%
% Design intent (from previous publication):
%   - Japanese body text in gothic (HaranoAjiGothic), bold = W6 weight
%   - Latin text in sans-serif (TeX Gyre Heros ≈ Helvetica)
%   - Chapter headings: 「第 X 章」 with thick grey rule below
%   - Section headings: grey rule below each level
%   - Images capped at line width × 0.9, max height 8 cm
%   - Caption without colon separator
%   - Section numbering depth: 4 (chapter→subsubsection)
% ====================================================================

% ---- Japanese engine ------------------------------------------------
\usepackage{luatexja}
\usepackage{luatexja-fontspec}

% ---- Japanese fonts: Harano Aji Gothic (= modern ゴシック体) --------
\setmainjfont{HaranoAjiGothic}[
  UprightFont = HaranoAjiGothic-Regular,
  BoldFont    = HaranoAjiGothic-Bold
]
\setsansjfont{HaranoAjiGothic}[
  UprightFont = HaranoAjiGothic-Regular,
  BoldFont    = HaranoAjiGothic-Bold
]

% ---- Latin fonts: TeX Gyre Heros (≈ Helvetica) ---------------------
\usepackage{fontspec}
\setmainfont{TeX Gyre Heros}[
  UprightFont = *-Regular,
  BoldFont    = *-Bold,
  ItalicFont  = *-Italic,
  BoldItalicFont = *-BoldItalic
]
\setsansfont{TeX Gyre Heros}[
  UprightFont = *-Regular,
  BoldFont    = *-Bold,
  ItalicFont  = *-Italic,
  BoldItalicFont = *-BoldItalic
]

% ---- Colours --------------------------------------------------------
\usepackage{xcolor}
\definecolor{mgrey}{gray}{0.75}
\definecolor{codebg}{RGB}{245,245,245}
\definecolor{codeframe}{RGB}{204,204,204}
\definecolor{commentgreen}{RGB}{0,128,0}
\definecolor{stringcolor}{RGB}{196,26,22}
\definecolor{keywordblue}{RGB}{0,0,255}

% ---- Section numbering depth ----------------------------------------
\setcounter{secnumdepth}{4}

% ---- Chapter / Section formatting (grey rules) ----------------------
\usepackage[bf, raggedright, compact]{titlesec}

% Chapter: 「第 1 章」 right-aligned, huge title, thick grey rule
\titleformat{\chapter}[display]
  {\large\bfseries}
  {\raggedleft 第 \thechapter{} 章}
  {1pt}
  {\huge\bfseries\raggedleft}
  [\color{mgrey}{\titlerule[4pt]}]

% Section: grey 2pt rule
\titleformat{\section}[hang]
  {\Large\bfseries}{\thesection}{12pt}{}
  [\color{mgrey}{\titlerule[2pt]}]

% Subsection: grey 1pt rule
\titleformat{\subsection}[hang]
  {\large\bfseries}{\thesubsection}{12pt}{}
  [\color{mgrey}{\titlerule[1pt]}]

% Subsubsection: grey 1pt rule
\titleformat{\subsubsection}[hang]
  {\normalsize\bfseries}{\thesubsubsection}{12pt}{}
  [\color{mgrey}{\titlerule[1pt]}]

% Paragraph: run-in
\titleformat{\paragraph}[runin]
  {\normalsize\bfseries}{\theparagraph}{12pt}{}[]

% ---- Spacing --------------------------------------------------------
\usepackage{setspace}
\setstretch{1.2}

\setlength{\parindent}{1em}
\setlength{\parskip}{0.5em}

% ---- Graphics – cap size to avoid overflow --------------------------
\usepackage{graphicx}
\makeatletter
\def\maxwidth{%
  \ifdim\Gin@nat@width>\linewidth 0.9\linewidth\else\Gin@nat@width\fi}
\def\maxheight{8cm}
\makeatother
\setkeys{Gin}{width=\maxwidth,height=\maxheight,keepaspectratio}

\usepackage{float}
\floatplacement{figure}{htbp}

% ---- Caption: remove colon separator --------------------------------
\usepackage{caption}
\captionsetup{labelsep=space}

% ---- Code listings ---------------------------------------------------
\usepackage{listings}

\lstset{
  basicstyle=\ttfamily\small,
  backgroundcolor=\color{codebg},
  frame=single,
  frameround=tttt,
  rulecolor=\color{codeframe},
  framerule=1.5pt,
  commentstyle=\color{commentgreen},
  stringstyle=\color{stringcolor},
  keywordstyle=\color{keywordblue}\bfseries,
  breaklines=true,
  breakatwhitespace=false,
  showstringspaces=false,
  tabsize=2,
  numbers=left,
  numberstyle=\tiny\color{gray},
  stepnumber=1,
  xleftmargin=2em,
  framexleftmargin=1.5em
}

% ---- Tables ----------------------------------------------------------
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{array}

% ---- URLs ------------------------------------------------------------
\usepackage{url}
\urlstyle{same}

% ---- Hyperlinks (Pandoc loads hyperref; configure after load) --------
\AtBeginDocument{
  \hypersetup{
    unicode=true,
    bookmarksnumbered=true,
    bookmarksopen=true,
    bookmarksopenlevel=2,
    colorlinks=true,
    linkcolor=red,
    citecolor=green,
    urlcolor=magenta,
    pdfstartview={FitH},
    pdfpagemode=UseOutlines
  }
}

% ---- Headers / Footers -----------------------------------------------
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[LE,RO]{\thepage}
\fancyhead[LO]{\nouppercase{\rightmark}}
\fancyhead[RE]{\nouppercase{\leftmark}}
\renewcommand{\headrulewidth}{0.4pt}

% ---- Orphan / Widow control ------------------------------------------
\widowpenalty=10000
\clubpenalty=10000
\raggedbottom

% ---- Abstract environment for book class -----------------------------
\newenvironment{abstract}{%
  \small
  \begin{center}\bfseries 概要\end{center}
  \quotation
}{\endquotation}

% ---- Technical term macros -------------------------------------------
\newcommand{\techterm}[1]{\texttt{\textbf{#1}}}
\newcommand{\BLE}{\techterm{BLE}}
\newcommand{\GATT}{\techterm{GATT}}
\newcommand{\UUID}{\techterm{UUID}}
\newcommand{\ATT}{\techterm{ATT}}
